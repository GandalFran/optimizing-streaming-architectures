### Kafka Topic Configuration

```yaml
apiVersion: kafka.strimzi.io/v1beta1
kind: KafkaTopic
metadata:
  name: payments
spec:
  partitions: 16
  replicas: 3
  config:
    retention.ms: 604800000
    segment.ms: 604800000
    cleanup.policy: compact
    flush.ms: 10000
    min.insync.replicas: 2
    acks: all
    replication.factor: 3
---
```

  name: flink-sql-job

  replicas: 3

    matchLabels:
      app: flink-sql-job

    metadata:
      labels:
        app: flink-sql-job

      containers:
