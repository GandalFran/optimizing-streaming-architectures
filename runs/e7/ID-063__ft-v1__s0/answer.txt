1. ```bash
# Create sensor-readings topic
kafka-topics.sh --create --topic sensor-readings --partitions 12 --replication-factor 3 --config retention.ms=604800000 --config retention.bytes=-1 --config min.insync.replicas=2 --bootstrap-server kafka:9092

# Create sensor-readings-clean topic
kafka-topics.sh --create --topic sensor-readings-clean --partitions 12 --replication-factor 3 --config retention.ms=604800000 --config retention.bytes=-1 --config min.insync.replicas=2 --bootstrap-server kafka:9092
```

2. ```yaml
# Flink configuration
jobmanager.rpc.address: jobmanager
jobmanager.rpc.port: 6123
jobmanager.web.port: 8081
taskmanager.numberOfTaskSlots: 4
state.backend: filesystem
state.backend.fs.checkpointdir: s3://flink-state/checkpoints
parallelism.default: 3
```

3. ```sql
# Flink SQL statements
CREATE TABLE sensor_readings (
  sensor_id STRING,
  event_time TIMESTAMP(3),
  temperature DOUBLE,
  humidity DOUBLE,
  PRIMARY KEY (sensor_id) NOT ENFORCED
) WITH (
  'connector' = 'kafka',
  'topic' = 'sensor-readings',
  'properties.bootstrap.servers' = 'kafka:9092',
  'format' = 'json',
  'scan.startup.mode' = 'latest-offset',
  'sink.partition-commit.policy.type' = 'all-to-same'
);

CREATE TABLE sensor_readings_clean (
  sensor_id STRING,
  event_time TIMESTAMP(3),
  temperature DOUBLE,
  humidity DOUBLE,
  PRIMARY KEY (sensor_id) NOT ENFORCED
) WITH (
  'connector' = 'kafka',
  'topic' = 'sensor-readings-clean',
  'properties.bootstrap.servers' = 'kafka:9092',
  'format' = 'json',
  'sink.partitioner.type' = 'round-robin',
  'sink.partition-commit.policy.type' = 'all-to-same'
);
```
