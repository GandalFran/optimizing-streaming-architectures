# 1. `kafka-topics.sh` commands to create the Kafka topics
bash
# create topic 'clicks'
kafka-topics.sh --create --topic clicks \
    --partitions 8 --replication-factor 3 \
    --config retention.ms=2592000000 --config cleanup.policy=compact \
    --bootstrap-server kafka:9092

# create topic 'clicks-enriched'
kafka-topics.sh --create --topic clicks-enriched \
    --partitions 8 --replication-factor 3 \
    --config retention.ms=2592000000 --config cleanup.policy=compact \
    --bootstrap-server kafka:9092
```
