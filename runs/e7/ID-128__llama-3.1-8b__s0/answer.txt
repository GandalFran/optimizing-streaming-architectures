kind: KafkaTopic

  name: clicks

  replicas: 3
  config:

    segment.ms: 604800000
    cleanup.policy: compact
    flush.ms: 10000

  name: flink-sql-job
spec:
  replicas: 1
  selector:
    matchLabels:
      app: flink-sql-job
  template:
    metadata:
      labels:
        app: flink-sql-job
    spec:
      containers:
      - name: flink
        image: flink:latest
        ports:
        - containerPort: 6123
        volumeMounts:
        - name: flink-conf
          mountPath: /etc/flink/conf
        - name: usrlib
          mountPath: /opt/flink/usrlib
        - name: checkpoints
          mountPath: /opt/flink/checkpoints
      volumes:
      - name: flink-conf
        configMap:
          name: flink-conf
      - name: usrlib

      - name: checkpoints

          claimName: flink-checkpoints
