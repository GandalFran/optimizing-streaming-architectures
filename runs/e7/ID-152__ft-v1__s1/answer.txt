# 1. Strimzi KafkaTopic resources for all topics

apiVersion: kafka.strimzi.io/v1beta1
kind: KafkaTopic
metadata:
  name: payments
spec:
  partitions: 3
  replicas: 3
  topicNameFormat: "${cluster.name}-${topic.name}"
  config:
    message.format.version: "2.5-rc1"
    min.insync.replicas: "1"
    "cleanup.policy": "delete"
    "delete.retention.ms": "2000"
    "retention.ms": "604800000"
    "compression.type": "lz4"
    "unclean.leader.election.enable": "true"
    "auto.create.topics.enable": "false"
    "auto.leader.rebalance.enable": "true"
    "auto.expire.unclean.leader.election.enable": "true"

apiVersion: kafka.strimzi.io/v1beta1
kind: KafkaTopic
metadata:
  name: payments-validated
spec:
  partitions: 3
  replicas: 3
  topicNameFormat: "${cluster.name}-${topic.name}"
  config:
    message.format.version: "2.5-rc1"
    min.insync.replicas: "1"
    "cleanup.policy": "delete"
    "delete.retention.ms": "2000"
    "retention.ms": "604800000"
    "compression.type": "lz4"
    "unclean.leader.election.enable": "true"
    "auto.create.topics.enable": "false"
    "auto.leader.rebalance.enable": "true"
    "auto.expire.unclean.leader.election.enable": "true"

# Justification: Each Kafka topic has 3 partitions and 3 replicas. The retention is set to 7 days, and the cleanup policy is set to delete. The min.insync.replicas is set to 1 to ensure that data is not lost in case of a broker failure. The compression type is set to lz4 to reduce the amount of data that is stored in the Kafka topics. The unclean.leader.election.enable is set to true to ensure that a broker can be elected as a leader even if it is not in sync with the other brokers.
