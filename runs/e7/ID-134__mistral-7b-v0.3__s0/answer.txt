 1. Strimzi `KafkaTopic` resources for all topics:

```yaml
apiVersion: kafka.strimzi.io/v1beta2
kind: Kafka
metadata:
  name: my-cluster
spec:
  kafka:
    replicas: 3
    listeners:
    - name: plain
      port: 9092
      tls: false
      config:
        advertisingListenerEnabled: "true"
        listenerType: "PLAINTEXT"
        listenerSecurityPolicy: "PLAINTEXT_LISTENER"
        zookeeperConnect: "my-cluster-zookeeper:2181"
        bootstrapServer: "my-cluster-kafka-bootstrap:9092"

---
apiVersion: kafka.strimzi.io/v1beta2
kind: KafkaTopic
metadata:
  name: vehicle-positions
spec:
  replicas: 3
  partitions: 10
  config:
    retention.ms: 86400000
    segment.bytes: 10485760
    compression.type: "lz4"

---
apiVersion: kafka.strimzi.io/v1beta2
kind: KafkaTopic
metadata:
  name: vehicle-positions-clean
spec:
  replicas: 3
  partitions: 10
  config:
    retention.ms: 86400000
    segment.bytes: 10485760
    compression.type: "lz4"
```
