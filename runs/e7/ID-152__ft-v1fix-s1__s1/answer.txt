# 1. Strimzi KafkaTopic resources for all topics

apiVersion: kafka.strimzi.io/v1beta1
kind: KafkaTopic
metadata:
  name: payments
spec:
  partitions: 3
  replicas: 3
  config:
    retention.ms: 604800000
  storage:
    class: kubernetes
    size: 10Gi
    volumeName: payments-storage
  topicNameFormat: payments-{{ . }}
  bootstrapServers: my-cluster-kafka-bootstrap:9092
  partitionsPerTopic: 3
  replicasPerPartition: 3
  retention: 7 days
  retentionBytes: 16800000000
  cleanupPolicy: "delete"
  compactedTopic: false
  minInsyncReplicas: 2
  configOverrides:
    - name: min.insync.replicas
      value: "2"
