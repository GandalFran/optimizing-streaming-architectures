1. ```bash
# Create the Kafka topics
kafka-topics.sh --create --topic banking-transactions --partitions 12 --replication-factor 3 --config retention.ms=604800000 --config min.insync.replicas=2 --bootstrap-server kafka:9092
kafka-topics.sh --create --topic banking-transactions-validated --partitions 12 --replication-factor 3 --config retention.ms=604800000 --config min.insync.replicas=2 --bootstrap-server kafka:9092
```

2. ```yaml
# Flink configuration
jobmanager.rpc.address: jobmanager
jobmanager.rpc.port: 6123
jobmanager.web.port: 8081
taskmanager.numberOfTaskSlots: 4
state.backend: filesystem
state.backend.fs.checkpointdir: s3://flink-state/checkpoints
parallelism.default: 3
```

3. ```sql
# Flink SQL statements
CREATE TABLE banking_transactions (
  account_id STRING,
  transaction JSON
) WITH (
  'connector' = 'kafka',
  'topic' = 'banking-transactions',
  'properties.bootstrap.servers' = 'kafka:9092',
  'format' = 'json'
);

CREATE TABLE banking_transactions_validated (
  account_id STRING,
  transaction JSON
) WITH (
  'connector' = 'kafka',
  'topic' = 'banking-transactions-validated',
  'properties.bootstrap.servers' = 'kafka:9092',
  'format' = 'json'
);
```
