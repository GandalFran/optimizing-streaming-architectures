# 1. kafka-topics.sh commands
# Create the Kafka topics
kafka-topics.sh --create --topic ad-impressions --partitions 10 --replication-factor 3 --config retention.ms=2592000000 --config cleanup.policy=compact --bootstrap-server kafka:9092
kafka-topics.sh --create --topic ad-impressions-enriched --partitions 10 --replication-factor 3 --config retention.ms=2592000000 --config cleanup.policy=compact --bootstrap-server kafka:9092

# Justify: 10 partitions are chosen to handle the input rate of 50,000 events/s. The replication factor is set to 3 to ensure high availability. Retention is set to 30 days, and the cleanup policy is set to compact to save space.

# 2. config.yaml
# Flink configuration
taskmanager.memory.process.size: 8000m
jobmanager.memory.process.size: 8000m
taskmanager.numberOfTaskSlots: 4
state.backend: filesystem
state.backend.fs.checkpointdir: s3://flink-state/checkpoints
parallelism.default: 3

# Justify: 8 GB of memory is allocated to each TaskManager, and 4 CPU cores are used. The state backend is set to filesystem, and the checkpoint storage is set to S3. The default parallelism is set to 3.

# 3. Flink SQL CREATE TABLE statements
CREATE TABLE ad_impressions (
  campaign_id STRING,
  event JSON
) WITH (
  'connector' = 'kafka',
  'topic' = 'ad-impressions',
  'properties.bootstrap.servers' = 'kafka:9092',
  'format' = 'json'
);

CREATE TABLE ad_impressions_enriched (
  campaign_id STRING,
  event JSON
) WITH (
  'connector' = 'kafka',
  'topic' = 'ad-impressions-enriched',
  'properties.bootstrap.servers' = 'kafka:9092',
  'format' = 'json'
);
