# 1. kafka-topics.sh commands
# Create the session-events topic
kafka-topics.sh --create --topic session-events --partitions 10 --replication-factor 3 --config retention.ms=604800000 --config retention.bytes=10737418240 --bootstrap-server kafka:9092

# Create the session-events-enriched topic
kafka-topics.sh --create --topic session-events-enriched --partitions 10 --replication-factor 3 --config retention.ms=604800000 --config retention.bytes=10737418240 --bootstrap-server kafka:9092

# Justification: 10 partitions per topic to sustain the input rate, 3 replicas to ensure durability, and 7 days of retention to meet the retention requirement.

# 2. config.yaml
# Set the parallelism to 10 to match the number of partitions
parallelism: 10

# Set the checkpointing interval to 10 seconds to ensure at-least-once delivery

execution.checkpointing.interval: 10000

# Justification: 10 seconds checkpointing interval to ensure at-least-once delivery, and 60 minutes retention to meet the retention requirement.

  user_id STRING,
  event JSON
