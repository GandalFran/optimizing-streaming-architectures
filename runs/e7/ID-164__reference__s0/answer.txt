```yaml
apiVersion: flink.apache.org/v1beta1
kind: FlinkDeployment
metadata:
  name: clicks-enriched
spec:
  image: flink-sql-runner:1.20  # flink:1.20 with sql-runner.jar and the SQL script in /opt/flink/usrlib
  flinkVersion: v1_20
  serviceAccount: flink
  flinkConfiguration:
    execution.checkpointing.dir: "s3://flink-state/checkpoints"
    execution.checkpointing.interval: "500 ms"
    execution.checkpointing.mode: "EXACTLY_ONCE"
    taskmanager.numberOfTaskSlots: "4"
    execution.checkpointing.savepoint-dir: "s3://flink-state/savepoints"
  jobManager:
    resource:
      memory: "2048m"
      cpu: 1
  taskManager:
    resource:
      memory: "8Gi"
      cpu: 4
  job:
    jarURI: local:///opt/flink/usrlib/sql-runner.jar
    args: ["/opt/flink/usrlib/sql-scripts/pipeline.sql"]
    parallelism: 3
    upgradeMode: savepoint
```

```sql
CREATE TABLE src (
  `session_id` STRING,
  `payload` STRING,
  `event_time` TIMESTAMP(3)
) WITH (
  'connector' = 'kafka',
  'topic' = 'clicks',
  'properties.bootstrap.servers' = 'my-cluster-kafka-bootstrap:9092',
  'properties.group.id' = 'clicks-enriched-job',
  'scan.startup.mode' = 'group-offsets',
  'properties.auto.offset.reset' = 'earliest',
  'format' = 'json'
);

CREATE TABLE snk (
  `session_id` STRING,
  `payload` STRING,
  `event_time` TIMESTAMP(3),
  `enriched_at` TIMESTAMP(3)
) WITH (
  'connector' = 'kafka',
  'topic' = 'clicks-enriched',
  'properties.bootstrap.servers' = 'my-cluster-kafka-bootstrap:9092',
  'format' = 'json',
  'sink.delivery-guarantee' = 'exactly-once',
  'sink.transactional-id-prefix' = 'clicks-enriched-tx',
  'properties.transaction.timeout.ms' = '600000'
);
```

6 partitions: 50,000 events/s x 512 B = 25.60 MB/s at 5 MB/s per partition (and at least the parallelism); parallelism 3 at 20,000 events/s per source subtask; replication factor 3 with min.insync.replicas=2, so acknowledged writes survive the loss of one broker; retention 30 days. Exactly-once: checkpointing in EXACTLY_ONCE mode with an interval of half the latency objective (read_committed consumers see results only when the checkpoint commits the Kafka transaction, so the checkpoint itself must fit in the objective), a transactional sink with a transactional-id prefix and a transaction timeout above the checkpoint interval and below the broker maximum of 15 minutes.
