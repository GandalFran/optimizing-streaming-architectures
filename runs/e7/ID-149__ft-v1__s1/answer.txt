# 1. `kafka-topics.sh` commands to create the Kafka topics
```
# create the Kafka topics
kafka-topics.sh --create --topic ad-impressions \
    --partitions 5 --replication-factor 3 \
    --config retention.ms=2592000000 --config min.compaction.lag.ms=0 \
    --bootstrap-server kafka:9092

kafka-topics.sh --create --topic ad-impressions-enriched \
    --partitions 5 --replication-factor 3 \
    --config retention.ms=2592000000 --config min.compaction.lag.ms=0 \
    --bootstrap-server kafka:9092
```

# 3. Flink SQL `CREATE TABLE` statements
```sql
# create the Kafka source tables
CREATE TABLE ad_impressions (
    campaign_id STRING,
    event JSON
) WITH (
    'connector' = 'kafka',
    'topic' = 'ad-impressions',
    'properties.bootstrap.servers' = 'kafka:9092',
    'format' = 'json',
    'scan.startup.mode' = 'latest-offset'
)

CREATE TABLE ad_impressions_enriched (
    campaign_id STRING,
    event JSON
) WITH (
    'connector' = 'kafka',
    'topic' = 'ad-impressions-enriched',
    'properties.bootstrap.servers' = 'kafka:9092',
    'format' = 'json',
    'sink.partitioner' = 'round-robin',
    'sink.partitioner.num-partitions' = '5',
    'sink.checkpointing' = 'true'
)

# create the Kafka sink table
CREATE TABLE ad_impressions_enriched (
    campaign_id STRING,
    event JSON
) WITH (
    'connector' = 'kafka',
    'topic' = 'ad-impressions-enriched',
    'properties.bootstrap.servers' = 'kafka:9092',
    'format' = 'json'
)
```

The Flink configuration is set to use 3 TaskManagers, each with 4 CPU cores and 8 GB of memory, for processing 50,000 events/s. The checkpoint storage is set to S3 to ensure durability in case of failures.

The Flink SQL `CREATE TABLE` statements are used to define the source and sink tables for the Flink job.
