 1. Strimzi `KafkaTopic` resources for all topics:

```yaml
apiVersion: kafka.strimzi.io/v1beta2
kind: Kafka
metadata:
  name: my-cluster
spec:
  kafka:
    replicas: 3
    listeners:
    - name: plain
      port: 9092
      tls: false
      config:
        advertisingListenerEnabled: "true"
        listenerType: "PLAINTEXT"
        listenerSecurityPolicy: "PLAINTEXT_LISTENER"
        zookeeperConnect: "my-cluster-zookeeper:2181"
    kafkaRaft:
      electionTimeoutMs: 15000
      heartbeatIntervalMs: 3000
    entityOperator:
      topicOperator:
        topic:
        - banking-transactions
          partitions: 10
          replicas: 3
          config:
            retention.ms: 604800000
            compression.type: "lz4"
            message.max.bytes: 1048576
        - banking-transactions-validated
          partitions: 10
          replicas: 3
          config:
            retention.ms: 604800000
            compression.type: "lz4"
            message.max.bytes: 1048576
```
