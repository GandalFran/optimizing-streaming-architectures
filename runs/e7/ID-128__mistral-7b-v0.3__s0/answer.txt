 1. Strimzi `KafkaTopic` resources for all topics:

```yaml
apiVersion: kafka.strimzi.io/v1beta2
kind: Kafka
metadata:
  name: my-cluster
spec:
  kafka:
    replicas: 3
    listeners:
    - name: plain
      port: 9092
      tls: false
      config:
        advertisingListenerEnabled: "true"
        listenerType: CONNECT
        listenerName: PLAIN
    zooKeeper:
      enabled: false
    entityOperator:
      topicOperator:
        create:
          - apiVersion: kafka.strimzi.io/v1beta2
            kind: KafkaTopic
            metadata:
              name: clicks
            spec:
              partitions: 10
              replicas: 3
              config:
                retention.ms: 604800000
                compression.type: lzo
                message.max.bytes: 1048576
                segment.bytes: 1073741824
                segment.ms: 3600000
                min.insync.replicas: 2
                unclean.leader.election.enable: "false"
                transaction.timeout.ms: 60000
                transaction.state.log.replication.factor: 3
                transaction.state.log.min.isr: 2

          - apiVersion: kafka.strimzi.io/v1beta2
            kind: KafkaTopic
            metadata:
              name: clicks-enriched
            spec:
              partitions: 10
              replicas: 3
              config:
                retention.ms: 604800000
                compression.type: lzo
                message.max.bytes: 1048576
                segment.bytes: 1073741824
                segment.ms: 3600000
                min.insync.replicas: 2
                unclean.leader.election.enable: "false"
                transaction.timeout.ms: 60000
                transaction.state.log.replication.factor: 3
                transaction.state.log.min.isr: 2
```
