# 1. `kafka-topics.sh` commands to create the Kafka topics
bash
# create sensor-readings topic
kafka-topics.sh --create --topic sensor-readings --partitions 8 --replication-factor 3 --config retention.ms=604800000 --config delete.retention.ms=604800000 --config min.insync.replicas=3 --bootstrap-server kafka:9092

# create sensor-readings-clean topic
kafka-topics.sh --create --topic sensor-readings-clean --partitions 8 --replication-factor 3 --config retention.ms=604800000 --config delete.retention.ms=604800000 --config min.insync.replicas=3 --bootstrap-server kafka:9092
# 

# 2. Flink configuration (`config.yaml` keys)
yaml
# ensure checkpointing is enabled and is durable
jobmanager.rpc.port: 6123
jobmanager.heap.size: 4000m
taskmanager.heap.size: 4000m
taskmanager.numberOfTaskSlots: 4

# checkpointing
state.backend: local
state.backend.fs.checkpointdir: s3://flink-state/checkpoints
job.checkpointing.mode: exactly-once
job.checkpointing.type: internal
job.checkpointing.interval: 30000
job.checkpointing.externalized.checkpoint.cleanup: retain-at-least-last-successful

# parallelism
jobmanager.rpc.address: "jobmanager-hostname"
taskmanager.numberOfTaskSlots: 4
parallelism.default: 4
taskmanager.memory.fraction: 0.75
jobmanager.memory.fraction: 0.75

# network buffer sizes
network_buffer_size: 16m
# 

# 3. Flink SQL `CREATE TABLE` statements
sql
CREATE TABLE sensor_readings (
  sensor_id STRING,
  reading STRING
) WITH (
  'connector' = 'kafka',
  'topic' = 'sensor-readings',
  'properties.bootstrap.servers' = 'kafka:9092',
  'format' = 'json',
  'scan.startup.mode' = 'latest-offset'
);

CREATE TABLE sensor_readings_clean (
  sensor_id STRING,
  reading STRING,
  derived_field STRING
) WITH (
  'connector' = 'kafka',
  'topic' = 'sensor-readings-clean',
  'properties.bootstrap.servers' = 'kafka:9092',
  'format' = 'json'
);
#
