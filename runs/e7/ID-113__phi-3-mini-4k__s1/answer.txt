 ```bash
# Creating Kafka topics
kafka-topics --bootstrap-server kafka:9092 --create --topic clicks --partitions 3 --replication-factor 3 --config retention.ms=2592000000
kafka-topics --bootstrap-server kafka:9092 --create --topic clicks-enriched --partitions 3 --replication-factor 3 --config retention.ms=2592000000
```

Justification:
- The `--partitions` and `--replication-factor` values are set to 3 to ensure high availability and fault tolerance in a KRaft mode cluster.
- The `--config retention.ms=2592000000` (30 days in milliseconds) ensures that data is retained for the required duration.

```sql
-- Flink SQL to create Kafka source and sink tables
CREATE TABLE clicks (
  session_id VARCHAR,
  event_timestamp TIMESTAMP,
  event_value STRUCT<json:STRING>
) WITH (
  'connector' = 'kafka',
  'topic' = 'clicks',
  'properties.bootstrap.servers' = 'kafka:9092',
  'scan.startup.mode' = 'earliest',
  'format.class' = 'io.confluent.connect.avro.AvroConverter',
  'key.converter.schema.registry.url' = 'http://schema-registry:8081',
  'value.converter.schema.registry.url' = 'http://schema-registry:8081',
  'value.converter' = 'io.confluent.connect.avro.AvroConverter',
  'key.converter' = 'io.confluent.connect.avro.AvroConverter',
  'value.deserializer' = 'io.confluent.connect.avro.AvroDataFormat$Value',
  'key.deserializer' = 'io.confluent.connect.avro.AvroDataFormat$Key'
) AS 'rawClicks';

CREATE TABLE clicks-enriched (
  session_id VARCHAR,
  event_timestamp TIMESTAMP,
  enriched_event_value STRUCT<json:STRING>
) WITH (
  'connector' = 'kafka',
  'topic' = 'clicks-enriched',
  'properties.bootstrap.servers' = 'kafka:9092',
  'format.class' = 'io.confluent.connect.avro.AvroConverter',
  'key.converter.schema.registry.url' = 'http://schema-registry:8081',
  'value.converter.schema.registry.url' = 'http://schema-registry:8081',
  'value.converter' = 'io.confluent.connect.avro.AvroConverter',
  'key.converter' = 'io.confluent.connect.avro.AvroConverter',
  'value.deserializer' = 'io.confluent.connect.avro.AvroDataFormat$Value',
  'key.deserializer' = 'io.confluent.connect.avro.AvroDataFormat$Key'
) AS 'enrichedClicks';
```

Justification:
- The SQL statements create Kafka source and sink tables with the necessary schema for the clickstream events.
- The Kafka connector configuration is set to use Avro serialization format for efficient data processing.
- The `scan.startup.mode` is set to 'earliest' to ensure that the earliest available events are consumed first.
- The schema registry URLs are provided to resolve Avro schemas during data processing.
