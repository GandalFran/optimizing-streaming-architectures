3. ```sql
# app-logs table
CREATE TABLE app-logs (
  service STRING,
  event STRING
) WITH (
  'connector' = 'kafka',
  'topic' = 'app-logs',
  'properties.bootstrap.servers' = 'kafka:9092',
  'format' = 'json',
  'scan.startup.mode' = 'latest-offset',
  'sink.partition-commit.policy.type' = 'all-complete'
)

# app-logs-parsed table
CREATE TABLE app-logs-parsed (
  service STRING,
  event STRING,
  derived_field STRING
) WITH (
  'connector' = 'kafka',
  'topic' = 'app-logs-parsed',
  'properties.bootstrap.servers' = 'kafka:9092',
  'format' = 'json',
  'sink.partitioner' = 'round-robin',
  'sink.partition-commit.policy.type' = 'all-complete'
)
```
