apiVersion: kafka.strimzi.io/v1beta1
kind: KafkaTopic
metadata:
  name: payments
spec:
  kafka:
    bootstrap:
      service:
        name: my-cluster-kafka-bootstrap
        port: 9092
    topic:
      partitions: 16
      replicationFactor: 3
      numPartitions: 16
      config:
        retention.ms: 604800000
        segment.ms: 604800000
        cleanup.policy: compact
        flush.ms: 10000
