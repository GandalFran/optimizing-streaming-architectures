# 1. `kafka-topics.sh` commands to create the Kafka topics
bash
# create the session events topic
kafka-topics.sh --create --topic session-events \
    --partitions 5 --replication-factor 3 \
    --config retention.ms=604800000 --config cleanup.policy=compact \
    --bootstrap-server kafka:9092

# create the enriched events topic
kafka-topics.sh --create --topic session-events-enriched \
    --partitions 5 --replication-factor 3 \
    --config retention.ms=604800000 --config cleanup.policy=compact \
    --bootstrap-server kafka:9092
```

# 2. Flink configuration
yaml
# set the parallelism for the Flink job
parallelism: 5

# set the checkpointing interval
checkpointing:
  interval: 5000
  mode: exact-once

# set the Flink state backend to use S3
state:
  backend: s3
  s3:
    path: s3://flink-state/checkpoints
```

  user_id STRING,
  event JSON
