 1. Strimzi `KafkaTopic` resources for all topics:

```yaml
apiVersion: kafka.strimzi.io/v1beta2
kind: Kafka
metadata:
  name: my-cluster
spec:
  kafka:
    replicas: 3
    listeners:
    - name: plain
      port: 9092
      tls: false
      config:
        advertisingListenerEnabled: "true"
        listenerType: "PLAINTEXT"
        listenerSecurityPolicy: "PLAINTEXT_LISTENER"
        zookeeperConnect: "zookeeper:2181"
        bootstrapServer: "my-cluster-kafka-bootstrap:9092"

---
apiVersion: kafka.strimzi.io/v1beta2
kind: KafkaTopic
metadata:
  name: clicks
spec:
  replicas: 3
  partitions: 12
  compaction:
    strategy: "compact"
  config:
    retention.ms: 2592000000
    segment.bytes: 1073741824
    cleanup.policy: "delete"
    min.insync.replicas: 2
    max.message.bytes: 1048576
    inter.broker.timeout.ms: 300

---
apiVersion: kafka.strimzi.io/v1beta2
kind: KafkaTopic

  name: clicks-enriched
```
