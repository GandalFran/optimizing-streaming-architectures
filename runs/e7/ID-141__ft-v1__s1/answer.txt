# 1. `kafka-topics.sh` commands to create the Kafka topics
bash
# create topic 'app-logs'
./bin/kafka-topics.sh --create --topic app-logs --partitions 48 --replication-factor 3 --config retention.ms=86400000 --config cleanup.policy=compact --bootstrap-server kafka:9092

# create topic 'app-logs-parsed'
./bin/kafka-topics.sh --create --topic app-logs-parsed --partitions 48 --replication-factor 3 --config retention.ms=86400000 --config cleanup.policy=compact --bootstrap-server kafka:9092
```
