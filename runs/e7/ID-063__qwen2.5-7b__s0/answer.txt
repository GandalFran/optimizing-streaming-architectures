```bash
#!/bin/bash
# Create Kafka topics with 3 partitions and 1 replica to ensure high availability and durability
kafka-topics.sh --create --topic sensor-readings --partitions 3 --replication-factor 1 --config retention.ms=604800000 --bootstrap-server kafka:9092
kafka-topics.sh --create --topic sensor-readings-clean --partitions 3 --replication-factor 1 --config retention.ms=604800000 --bootstrap-server kafka:9092
```
