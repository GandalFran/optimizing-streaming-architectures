```yaml
apiVersion: kafka.strimzi.io/v1beta2
kind: Kafka
metadata:
  name: my-cluster
spec:
  kafka:
    replicas: 3
    listeners:
    - name: plain
      port: 9092
      tls: false
      authentication: none
    zookeeper:
      enabled: true
      existingZookeeper:
        servers:
        - my-cluster-zookeeper:2181
    kafkaRaft:
      enabled: true

---

apiVersion: kafka.strimzi.io/v1beta2
kind: KafkaTopic
metadata:
  name: card-transactions
spec:
  replicas: 3
  partitions: 10
  compaction:
    strategy: compact
  config:
    retention.ms: 2592000000
    segment.bytes: 1073741824

    max.message.per.partition.per.epoch: 10000000
```
