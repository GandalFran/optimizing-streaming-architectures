### 3. Flink SQL `CREATE TABLE` statements

```sql
-- Create the source table
CREATE TABLE vehicle_positions (
  vehicle_id STRING,
  -- other columns...
  WATERMARK FOR vehicle_id AS PROCTIME() - INTERVAL '10' SECOND
) WITH (
  'connector' = 'kafka',
  'topic' = 'vehicle-positions',
  'properties.bootstrap.servers' = 'my-cluster-kafka-bootstrap:9092',
  'scan.startup.mode' = 'earliest',
  'properties.group.id' = 'flink-sql-job-group',
  'scan.intermediate.checkpoint.interval' = '10000',
  'scan.commit.interval' = '10000'
);

-- Create the sink table
CREATE TABLE vehicle_positions_clean (
  vehicle_id STRING,
  -- other columns...
) WITH (
  'connector' = 'kafka',
  'topic' = 'vehicle-positions-clean',
  'properties.bootstrap.servers' = 'my-cluster-kafka-bootstrap:9092',
  'properties.key.format' = 'json',
  'properties.value.format' = 'json',
  'properties.key.json.schema' = 'struct<vehicle_id:STRING>',
  'properties.value.json.schema' = 'struct<...>',
  'properties.producer.acks' = 'all',
  'properties.producer.retries' = '3',
  'properties.producer.acks.timeout' = '10000'
);

-- Create a function to filter out malformed events
CREATE FUNCTION filter_malformed_events AS 'com.example.MalformedEventFilter';

-- Create a function to enrich each event with derived fields
CREATE FUNCTION enrich_events AS 'com.example.EventEnricher';

-- Create the SQL job
CREATE TABLE enriched_events AS
  SELECT
    vehicle_id,
    -- other columns...
  FROM (
    SELECT
      vehicle_id,
      -- other columns...
    FROM vehicle_positions
    WHERE filter_malformed_events(vehicle_id, ...) = FALSE
  ) AS filtered_events
  LATERAL TABLE enrich_events(filtered_events) AS (
    -- derive additional fields here
  );

-- Write the enriched events to the sink table
INSERT INTO vehicle_positions_clean
SELECT * FROM enriched_events;
```

### Justification of main values:

*   The number of Kafka partitions (16) is chosen to allow for a balanced distribution of events across the brokers, considering the expected input rate of 20,000 events/s and the assumption that one Kafka partition can sustain at most 5 MB/s of produce traffic.
*   The retention time for both topics is set to 1 day (86,400,000 ms) to match the requirements.
*   The Flink configuration is set to use 4 CPU cores and 16 GiB of memory for the JobManager, and 4 CPU cores and 8 GiB of memory for each TaskManager. This allows for a good balance between resource utilization and overhead.
