```bash
kafka-topics.sh --bootstrap-server kafka:9092 --create --topic session-events --partitions 3 --replication-factor 3 \
  --config min.insync.replicas=2 \
  --config retention.ms=86400000
kafka-topics.sh --bootstrap-server kafka:9092 --create --topic session-events-enriched --partitions 3 --replication-factor 3 \
  --config min.insync.replicas=2 \
  --config retention.ms=86400000
```

```yaml
execution:
  checkpointing:
    dir: s3://flink-state/checkpoints
    interval: 10 s
    mode: AT_LEAST_ONCE
parallelism:
  default: 1
taskmanager:
  memory:
    process:
      size: 8g
  numberOfTaskSlots: 4
```

```sql
CREATE TABLE src (
  `user_id` STRING,
  `payload` STRING,
  `event_time` TIMESTAMP(3)
) WITH (
  'connector' = 'kafka',
  'topic' = 'session-events',
  'properties.bootstrap.servers' = 'kafka:9092',
  'properties.group.id' = 'session-events-enriched-job',
  'scan.startup.mode' = 'group-offsets',
  'properties.auto.offset.reset' = 'earliest',
  'format' = 'json'
);

CREATE TABLE snk (
  `user_id` STRING,
  `payload` STRING,
  `event_time` TIMESTAMP(3),
  `enriched_at` TIMESTAMP(3)
) WITH (
  'connector' = 'kafka',
  'topic' = 'session-events-enriched',
  'properties.bootstrap.servers' = 'kafka:9092',
  'format' = 'json',
  'sink.delivery-guarantee' = 'at-least-once'
);
```

3 partitions: 20,000 events/s x 512 B = 10.24 MB/s at 5 MB/s per partition (and at least the parallelism); parallelism 1 at 20,000 events/s per source subtask; replication factor 3 with min.insync.replicas=2, so acknowledged writes survive the loss of one broker; retention 1 days. At-least-once: checkpointing enabled so the sink flushes on every checkpoint.
