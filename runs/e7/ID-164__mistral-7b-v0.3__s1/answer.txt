 1. Strimzi `KafkaTopic` resources for all topics:

```yaml
apiVersion: kafka.strimzi.io/v1beta2
kind: Kafka
metadata:
  name: my-cluster
spec:
  kafka:
    replicas: 3
    listeners:
    - name: plain
      port: 9092
      tls: false
      advertising: true
      openCluster: true
    entityOperator:
      replicas: 3
    zookeeper:
      replicas: 3
      storageClass: standard
    kafkaStorage:
      volumeClaimTemplate:
        storageClassName: standard
        resources:
          requests:
            storage: 10Gi

---
apiVersion: kafka.strimzi.io/v1beta2
kind: KafkaTopic
metadata:
  name: clicks
spec:
  replicas: 3
  partitions: 10
  compaction: daily
  config:
    retention.ms: 2592000000  # 30 days in milliseconds
    segment.bytes: 1073741824  # 1 GB
    message.max.bytes: 1048576  # 1 MB
    min.insync.replicas: 2
    replication.factor: 3
```
