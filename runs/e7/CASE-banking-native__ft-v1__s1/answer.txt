# 1. `kafka-topics.sh` commands to create the Kafka topics
bash
# create banking transactions topic
kafka-topics.sh --create --topic banking-transactions \
    --partitions 5 --replication-factor 3 \
    --config retention.ms=604800000 --config cleanup.policy=compact \
    --bootstrap-server kafka:9092

# create validated transactions topic
kafka-topics.sh --create --topic banking-transactions-validated \
    --partitions 5 --replication-factor 3 \
    --config retention.ms=604800000 --config cleanup.policy=compact \
    --bootstrap-server kafka:9092
# 2. Flink configuration (`config.yaml`)
yaml
# set the parallelism for the Flink job
parallelism: 5

# set the checkpointing interval
checkpointing:
  enabled: true
  internal-state-ttl: 1h
  min-pause-between-checkpoints: 0s
  max-concurrent-checkpoints: 1
  external-state-ttl: 24h
  persistence: s3
  persistence.path: s3://flink-state/checkpoints

# set the serialization format for the events
event-time:
  serialization.format: avro
# 3. Flink SQL `CREATE TABLE` statements
sql
# source table
CREATE TABLE banking_transactions (
  account_id STRING,
  transaction_id STRING,
  amount DECIMAL(38, 8),
  currency STRING,
  created_at TIMESTAMP(3),
  WATERMARK FOR created_at AS created_at - INTERVAL '5' SECOND
) WITH (
  'connector' = 'kafka',
  'topic' = 'banking-transactions',
  'properties.bootstrap.servers' = 'kafka:9092',
  'format' = 'json',
  'format.ignore-parse-errors' = 'true',
  'format.fail-on-missing-field' = 'false',
  'topic.event-time' = 'created_at',
  'topic.ignore-list' = 'account_id',
  'connector.scan.start' = 'earliest'
)

# sink table
CREATE TABLE banking_transactions_validated (
  account_id STRING,
  transaction_id STRING,
  amount DECIMAL(38, 8),
  currency STRING,
  created_at TIMESTAMP(3)
) WITH (
  'connector' = 'kafka',
  'topic' = 'banking-transactions-validated',
  'properties.bootstrap.servers' = 'kafka:9092',
  'format' = 'json',
  'topic.event-time' = 'created_at',
  'topic.ignore-list' = 'account_id',
  'connector.scan.start' = 'earliest'
)
